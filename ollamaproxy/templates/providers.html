Jak to použít v aplikaci

from openai import OpenAI
client = OpenAI(base_url="{{ root }}/providers/openai/v1", api_key="opx_…")   # proxy klíč, ne klíč OpenAI

from anthropic import Anthropic
client = Anthropic(base_url="{{ root }}/providers/anthropic", api_key="opx_…")

# Open WebUI → Admin → Settings → Connections → OpenAI API:
